```auto
每个配置项的含义均在英文下方


accepts-transfers=false
#允许Transfer数据包登录(1.20.5+)

allow-flight=true
#允许飞行，开船从高空降落/网络卡顿/命令方块设定玩家状态有可能误判，建议保持开启

allow-nether=true
#启用下界，关闭后不生成下界并无法进入，高版本被替换为无法通过下界门进入配置项

broadcast-console-to-ops=true
#广播控制台输出到在线op，开启后终端输出会同步给在线op

broadcast-rcon-to-ops=true
#广播rcon输出到在线op，开启后rcon输出会同步给在线op

bug-report-link=
#这项用不上，不管

difficulty=hard
#游戏难度，可选peaceful（和平），easy（简单），normal（普通）和 hard（困难），等级低于游戏内使用命令/difficulty <难度>

enable-command-block=true
#是否启用命令方块，若用不上请关闭防止炸服

enable-jmx-monitoring=false
#是否启用公开部分服务器运行参数，请参考https://docs.oracle.com/javase/8/docs/technotes/guides/management/agent.html

enable-query=false
#是否启用查询,害怕被炸服请关闭查询

enable-rcon=false
#是否启用rcon远程命令行，极度危险，不建议开启

enable-status=true
#是否启用开机状态，让服务器在服务器列表中看起来是在线的，怕炸服的可以关闭

enforce-secure-profile=true
#强制安全配置文件，让没有Mojang签名公钥的玩家无法连接到服务器

enforce-whitelist=false
#强制白名单，请搭配正版验证使用

entity-broadcast-range-percentage=100
#实体广播范围百分比，有效值10-1000，值越大性能需求越高

force-gamemode=false
#强制游戏模式，玩家连入游戏时会被强制更改为设置的游戏模式，禁用则沿用退出时的游戏模式，启用应当改为对应模式英文名

function-permission-level=2
#函数权限等级，范围1-4

gamemode=survival
#游戏模式设定

generate-structures=true
#是否生成除地牢外的结构

generator-settings={}
#自定义世界生成设置，参考wiki，https://zh.minecraft.wiki/w/%E8%87%AA%E5%AE%9A%E4%B9%89%E4%B8%96%E7%95%8C%E7%94%9F%E6%88%90

hardcore=false
#硬核模式，玩家将处于极限模式(可以用指令调模式)

hide-online-players=false
#隐藏在线玩家，启用则状态请求时不会发送在线玩家列表，推荐始终保持true避免被社工学炸服

initial-disabled-packs=
#初始禁用的数据包，以逗号分隔

initial-enabled-packs=vanilla
#初始启用的数据包，以逗号分隔，默认值是原版数据

level-name=world
#世界名称/地图名称/存档名称

level-seed=
#生成世界/地图/存档所用的种子，默认空则随机
level-type=minecraft\:normal
       #用于确定世界生成的类型，使用世界预设 ID 时需要转义 “：”，并且可以省略原版世界预设 ID 的命名空间 minecraft:
       世界预设包含：
       minecraft:normal：正常的世界
       minecraft:flat：超平坦世界，可通过预设修改
       minecraft:large_biomes：放大生物群系
       minecraft:amplified：放大化世界：提升建筑高度限制
       minecraft:single_biome_surface：单一群系世界
       buffet：仅限1.15及以前的版本。如果不改动生成器设置则与normal相同
       default_1_1：仅限 1.15 及更早版本。与默认值相同，但计为不同的世界类型。
       customized ：自定义，仅限1.15及更早版本，1.13后与normal相同，1.12及以前可用于创建完全自定义的世界。关于自定义的详细信息
       链接：https://zh.minecraft.wiki/w/%E8%87%AA%E5%AE%9A%E4%B9%89/Java%E7%89%881.13%E5%89%8D

log-ips=true
#登入时记录IP，1.20.2+用此配置，挺方便ban-ip

max-chained-neighbor-updates=1000000
#限制连锁NC更新的数量，超过此数量的连锁NC更新会被跳过。若为负数则无限制

max-players=20
#服务器软件上允许的最大玩家数量，如果在线人数达到后将拒绝新玩家连入，op连入不受影响，极限值2147483647

max-tick-time=60000
#最大游戏

max-world-size=29999984
#最大地图尺寸

motd=xxxxxxxxxxxx
#服务器在列表里显示的信息，支持Unicode转义字符

network-compression-threshold=256
#网络压缩阈值，此处引用wiki。默认会允许n-1字节的数据包正常发送, 如果数据包为n字节或更大时会进行压缩。所以，更低的数值会使得更多的数据包被压缩，但是如果被压缩的数据包字节太小将反而使压缩后字节更大。 -1 ： 完全禁用数据包压缩 0 ： 压缩全部数据包 注：以太网规范要求把小于64字节的数据包填充为64字节。因此，设置一个低于64的值可能没有什么好处。也不推荐让设置的值超过MTU（通常为1500字节）。

online-mode=false
#在线模式，默认采用官方提供的身份验证服务器，容易宕机

op-permission-level=4
#op权限等级，不建议向除服管以外的玩家提供2以上的权限

pause-when-empty-seconds=60
#服务器空闲暂停阈值，当超过xx秒没有玩家时，暂停服务器以减少占用

player-idle-timeout=0
#挂机玩家踢出阈值，单位分钟，0为禁用

prevent-proxy-connections=false
#防止代理连接，阻止代理者连接服务器，在高风险环境下推荐开启

pvp=true
#启用玩家间伤害

query.port=25565
#监听开放端口，即游戏连接端口

rate-limit=0

rcon.password=

rcon.port=25575

region-file-compression=deflate

require-resource-pack=false

resource-pack=

resource-pack-id=

resource-pack-prompt=

resource-pack-sha1=

server-ip=0.0.0.0

server-port=25565

simulation-distance=10

spawn-monsters=true

spawn-protection=0

sync-chunk-writes=true

text-filtering-config=

text-filtering-version=0

use-native-transport=true

view-distance=10

white-list=false
```
